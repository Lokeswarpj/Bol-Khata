bolkhata_voice_stream.py 
from bolkhata import HandsFreeLedger, GnaniNeuralEngine

# Low-token streaming voice pipeline with local audio caching
engine = GnaniNeuralEngine(model="prisma-v3.0", tts_model="timbre-v2.5")
ledger = HandsFreeLedger(owner_id=1, wake_word="BolKhata")

async def on_counter_audio_frame(audio_stream):
    # Handles numbers under noise ("teen sau chaalis" / "dedh sau" / "340")
    intent = await ledger.parse_and_execute(audio_stream)
    if intent.is_debt_cleared:
        await engine.speak_confirmation(f"{intent.customer} ka hisaab poora chukta hua. ₹0 baaki.")
